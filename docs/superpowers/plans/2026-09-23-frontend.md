import axios from 'axios'
const http = axios.create({ baseURL: '/api' })
export const promptsApi = {
  list: (params) => http.get('/prompts', { params }),
  create: (d) => http.post('/prompts', d),
  update: (id, d) => http.put(`/prompts/${id}`, d),
  remove: (id) => http.delete(`/prompts/${id}`),

export const libsApi = {
  groups: (lib) => http.get(`/libraries/${lib}/groups`),
  createGroup: (lib, name) => http.post(`/libraries/${lib}/groups`, { name }),
  renameGroup: (lib, id, name) => http.put(`/libraries/${lib}/groups/${id}`, { name }),
  deleteGroup: (lib, id) => http.delete(`/libraries/${lib}/groups/${id}`),
  images: (lib, params) => http.get(`/libraries/${lib}/images`, { params }),
  upload: (lib, groupId, files) => {
    const fd = new FormData()
    fd.append('group_id', groupId)
    files.forEach(f => fd.append('files', f))
    return http.post(`/libraries/${lib}/images`, fd)
  },
  updateImage: (lib, id, d) => http.put(`/libraries/${lib}/images/${id}`, d),
  deleteImage: (lib, id) => http.delete(`/libraries/${lib}/images/${id}`),

export const providersApi = {
  list: () => http.get('/providers'),
  create: (d) => http.post('/providers', d),

export const tasksApi = {

  list: (params) => http.get('/tasks', { params }),
  detail: (id) => http.get(`/tasks/${id}`),

- [ ] **步骤 1：查找/替换组合式函数（硬需求，逻辑从原型移植）**

`src/composables/useFindReplace.js`：
```js
import { ref, computed } from 'vue'
export function useFindReplace(textRef) {
  const query = ref(''), replaceWith = ref(''), idx = ref(0), matches = ref([])
  const count = computed(() => matches.value.length)
  const label = computed(() => !query.value ? '0/0' : count.value ? `${idx.value + 1}/${count.value}` : '无结果')
  function rescan() {
    matches.value = []
    if (query.value) {
      const re = new RegExp(query.value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi')
      let m
      while ((m = re.exec(textRef.value)) !== null) {
        matches.value.push(m.index)
        if (m.index === re.lastIndex) re.lastIndex++
      }
    }
    idx.value = 0
  }
  function jump(dir, editor) {
    if (!count.value) return
    idx.value = (idx.value + dir + count.value) % count.value
    const pos = matches.value[idx.value]
    editor.value?.focus()
    editor.value?.setSelectionRange(pos, pos + query.value.length)
    const line = textRef.value.slice(0, pos).split('\n').length
    editor.value.scrollTop = Math.max(0, (line - 4) * 22)
  }
  function replaceOne() {
    if (!count.value) return
    const pos = matches.value[idx.value]
    textRef.value = textRef.value.slice(0, pos) + replaceWith.value +
                    textRef.value.slice(pos + query.value.length)
    rescan()
  }
  function replaceAll() {
    if (!query.value) return 0
    const n = count.value
    textRef.value = textRef.value.replace(
      new RegExp(query.value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'), 'gi'), replaceWith.value)
    rescan()
    return n
  }
  return { query, replaceWith, label, rescan, jump, replaceOne, replaceAll }
}
```

- [ ] **步骤 1：轮询组合式函数**

```js
import { onMounted, onUnmounted } from 'vue'
import { tasksApi } from '../api'
export function useTaskPolling(refresh, hasActive) {
  let timer = null
  async function tick() {
    if (hasActive()) await refresh(false)
  }
  onMounted(() => { timer = setInterval(tick, 2000); tick() })
  onUnmounted(() => clearInterval(timer))
}
```
（`hasActive` 为「列表中存在 queued/running」的判断；refresh 静默拉列表。）
